      <h3 class="text-lg font-bold text-rose-700">PERINGATAN KECURANGAN!</h3>
      <p class="text-xs text-slate-600 leading-relaxed">
        Sistem mendeteksi bahwa Anda telah berpindah tab/layar! Pelanggaran ini dicatat secara otomatis ke dalam database guru.
      </p>
      <div class="p-2 bg-rose-50 text-rose-700 text-xs font-bold rounded-lg">
        Jumlah Pelanggaran: <span id="cheat-count">0</span> Kali

      <button onclick="closeCheatModal()" class="w-full bg-rose-600 hover:bg-rose-700 text-white font-bold py-2.5 rounded-lg text-sm transition">

  <!-- CLIENT SCRIPT -->
  <script>
    // Global State
    let studentInfo = null;
    let examQuestions = [];
    let studentAnswers = {};
    let raguAnswers = {};
    let currentQuestionIdx = 0;
    let examTimerInterval = null;
    let secondsRemaining = <?= config.DURATION_MINUTES ?> * 60;
    let cheatViolationCount = 0;

    // Local Storage Autosave Key
    const AUTOSAVE_KEY = "CBT_ANSWERS_FIQIH_V1";

    // App Initialization
    window.addEventListener('load', () => {
      blockDevToolsAndCopy();
      setupAntiCheat();
    });

    // Role Switch Login
    function switchLoginRole(role) {
      const btnSiswa = document.getElementById('btn-tab-siswa');
      const btnAdmin = document.getElementById('btn-tab-admin');
      const formSiswa = document.getElementById('form-login-siswa');
      const formAdmin = document.getElementById('form-login-admin');

      if (role === 'siswa') {
        btnSiswa.className = "flex-1 py-3 text-sm font-semibold text-emerald-700 border-b-2 border-emerald-600 focus:outline-none transition";
        btnAdmin.className = "flex-1 py-3 text-sm font-semibold text-slate-500 hover:text-emerald-700 focus:outline-none transition";
        formSiswa.classList.remove('hidden');
        formAdmin.classList.add('hidden');
      } else {
        btnAdmin.className = "flex-1 py-3 text-sm font-semibold text-emerald-700 border-b-2 border-emerald-600 focus:outline-none transition";
        btnSiswa.className = "flex-1 py-3 text-sm font-semibold text-slate-500 hover:text-emerald-700 focus:outline-none transition";
        formAdmin.classList.remove('hidden');
        formSiswa.classList.add('hidden');
      }
    }

    // Student Login
    function handleStudentLogin() {
      const nama = document.getElementById('login-nama').value;
      const token = document.getElementById('login-token').value;
      const alertBox = document.getElementById('login-alert');
      const alertMsg = document.getElementById('login-alert-msg');

      if (!nama || !token) {
        alertMsg.innerText = "Harap isi Nama dan Token Akses!";
        alertBox.classList.remove('hidden');
        return;
      }

      alertBox.classList.add('hidden');
      google.script.run.withSuccessHandler(res => {
        if (res.success) {
          studentInfo = res.siswa;
          startStudentExam();
        } else {
          alertMsg.innerText = res.message;
          alertBox.classList.remove('hidden');
        }
      }).verifyStudentLogin(nama, token);
    }

    // Admin Login
    function handleAdminLogin() {
      const pass = document.getElementById('login-admin-pass').value;
      const alertBox = document.getElementById('login-alert');
      const alertMsg = document.getElementById('login-alert-msg');

      google.script.run.withSuccessHandler(res => {
        if (res.success) {
          document.getElementById('view-login').classList.add('hidden');
          document.getElementById('view-admin-dashboard').classList.remove('hidden');
          updateLivePreview();
        } else {
          alertMsg.innerText = res.message;
          alertBox.classList.remove('hidden');
        }
      }).verifyAdminLogin(pass);
    }

    // Start Student Exam
    function startStudentExam() {
      document.getElementById('view-login').classList.add('hidden');
      document.getElementById('view-student-exam').classList.remove('hidden');
      document.getElementById('user-display-nama').innerText = studentInfo.nama;

      // Restore Local Storage Answers
      const saved = localStorage.getItem(AUTOSAVE_KEY);
      if (saved) {
        try { studentAnswers = JSON.parse(saved); } catch(e){}
      }

      // Fetch Questions
      google.script.run.withSuccessHandler(questions => {
        examQuestions = questions;
        renderQuestion(0);
        renderGridNavigator();
        startTimer();
      }).getQuestionsForStudent();
    }

    // Render Question Card
    function renderQuestion(idx) {
      if (idx < 0 || idx >= examQuestions.length) return;
      currentQuestionIdx = idx;
      const q = examQuestions[idx];

      document.getElementById('lbl-soal-nomor').innerText = idx + 1;
      document.getElementById('lbl-soal-bobot').innerText = q.bobot;
      document.getElementById('question-text-area').innerText = q.teksSoal;

      // Stimulus Panel Render
      const stimContent = document.getElementById('stimulus-content');
      if (q.teksStimulus || q.gambarStimulusUrl) {
        let stimHtml = '';
        if (q.gambarStimulusUrl) {
          stimHtml += `<img src="${q.gambarStimulusUrl}" class="rounded-lg mb-2 max-h-48 object-contain">`;
        }
        if (q.teksStimulus) {
          stimHtml += `<p class="whitespace-pre-line">${q.teksStimulus}</p>`;
        }
        stimContent.innerHTML = stimHtml;
      } else {
        stimContent.innerHTML = `<p class="text-slate-400 italic text-xs">Tidak ada bacaan/stimulus untuk soal ini.</p>`;
      }

      // Options Render
      const optionsContainer = document.getElementById('options-container');
      const opts = [
        { key: 'A', text: q.pilihanA },
        { key: 'B', text: q.pilihanB },
        { key: 'C', text: q.pilihanC },
        { key: 'D', text: q.pilihanD }
      ];

      const currentAns = studentAnswers[q.id] || "";

      optionsContainer.innerHTML = opts.map(o => {
        const isSelected = currentAns === o.key;
        const bgClass = isSelected 
          ? 'bg-emerald-50 border-emerald-600 text-emerald-900 font-semibold shadow-sm' 
          : 'bg-white border-slate-200 text-slate-700 hover:bg-slate-50';

        return `
          <div onclick="selectAnswer('${q.id}', '${o.key}')" class="p-3.5 rounded-xl border-2 ${bgClass} cursor-pointer transition flex items-start space-x-3">
            <div class="w-6 h-6 rounded-full border-2 border-emerald-600 flex items-center justify-center shrink-0 mt-0.5 ${isSelected ? 'bg-emerald-600 text-white' : 'bg-white text-emerald-600 font-bold'}">
              ${o.key}
            </div>
            <div class="text-sm leading-relaxed">${o.text}</div>
          </div>
        `;
      }).join('');

      // Nav Button Disables
      document.getElementById('btn-prev').disabled = (idx === 0);
      document.getElementById('btn-next').disabled = (idx === examQuestions.length - 1);
      
      // Update Ragu State
      const btnRagu = document.getElementById('btn-ragu');
      if (raguAnswers[q.id]) {
        btnRagu.className = "px-4 py-2 bg-amber-400 text-slate-900 font-bold rounded-lg text-xs md:text-sm flex items-center transition shadow";
      } else {
        btnRagu.className = "px-4 py-2 bg-amber-100 hover:bg-amber-200 text-amber-800 font-bold rounded-lg text-xs md:text-sm flex items-center transition";
      }

      renderGridNavigator();
    }

    // Select Answer
    function selectAnswer(qId, key) {
      studentAnswers[qId] = key;
      localStorage.setItem(AUTOSAVE_KEY, JSON.stringify(studentAnswers));
      renderQuestion(currentQuestionIdx);
    }

    // Toggle Ragu
    function toggleRaguRagu() {
      const q = examQuestions[currentQuestionIdx];
      raguAnswers[q.id] = !raguAnswers[q.id];
      renderQuestion(currentQuestionIdx);
    }

    // Question Navigation
    function navigateQuestion(step) {
      renderQuestion(currentQuestionIdx + step);
    }

    // Grid Navigator
    function renderGridNavigator() {
      const container = document.getElementById('grid-nav-container');
      container.innerHTML = examQuestions.map((q, i) => {
        const isCurrent = i === currentQuestionIdx;
        const isAnswered = !!studentAnswers[q.id];
        const isRagu = !!raguAnswers[q.id];

        let bg = "bg-slate-200 text-slate-700";
        if (isRagu) bg = "bg-amber-400 text-slate-900 font-bold";
        else if (isAnswered) bg = "bg-emerald-600 text-white font-bold";

        let ring = isCurrent ? "ring-2 ring-emerald-600 ring-offset-2" : "";

        return `
          <button onclick="renderQuestion(${i})" class="w-9 h-9 text-xs rounded-lg ${bg} ${ring} flex items-center justify-center transition">
            ${i + 1}
          </button>
        `;
      }).join('');
    }

    // Countdown Timer
    function startTimer() {
      const timerDisplay = document.getElementById('timer-countdown');
      examTimerInterval = setInterval(() => {
        secondsRemaining--;
        const mins = Math.floor(secondsRemaining / 60);
        const secs = secondsRemaining % 60;
        timerDisplay.innerText = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

        if (secondsRemaining <= 0) {
          clearInterval(examTimerInterval);
          alert("Waktu ujian telah habis! Jawaban Anda akan dikirim otomatis.");
          submitExam();
        }
      }, 1000);
    }

    // Mobile Drawer Toggle
    function toggleMobilePanel(side) {
      if (side === 'left') {
        const p = document.getElementById('panel-stimulus');
        p.classList.toggle('hidden');
      } else {
        const p = document.getElementById('panel-navigator');
        p.classList.toggle('hidden');
      }
    }

    // Confirm Finish Exam
    function confirmFinishExam() {
      if (confirm("Apakah Anda yakin ingin menyelesaikan dan mengirimkan jawaban ujian ini?")) {
        submitExam();
      }
    }

    // Submit Exam to Backend
    function submitExam() {
      clearInterval(examTimerInterval);
      const payload = {
        nama: studentInfo.nama,
        kelas: studentInfo.kelas,
        jawabanSiswa: studentAnswers,
        pelanggaranTab: cheatViolationCount
      };

      google.script.run.withSuccessHandler(res => {
        if (res.success) {
          localStorage.removeItem(AUTOSAVE_KEY);
          document.getElementById('res-score').innerText = res.nilaiAkhir;
          document.getElementById('res-benar').innerText = res.totalBenar;
          document.getElementById('res-salah').innerText = res.totalSalah;

          // Render Pembahasan
          const listContainer = document.getElementById('pembahasan-list');
          listContainer.innerHTML = res.rekapPembahasan.map(p => `
            <div class="p-3 rounded-lg border text-xs ${p.isCorrect ? 'bg-emerald-50 border-emerald-200' : 'bg-rose-50 border-rose-200'}">
              <div class="font-bold text-slate-800">Soal #${p.id}: ${p.teksSoal}</div>
              <div class="mt-1 flex space-x-4">
                <span>Jawaban Anda: <strong class="${p.isCorrect ? 'text-emerald-700' : 'text-rose-700'}">${p.jawabanSiswa}</strong></span>
                <span>Kunci: <strong class="text-emerald-700">${p.kunciJawaban}</strong></span>
              </div>
              <div class="mt-1 text-slate-600 italic">Pembahasan: ${p.pembahasan}</div>
            </div>
          `).join('');

          document.getElementById('modal-result').classList.remove('hidden');
        } else {
          alert("Gagal mengirim jawaban: " + res.message);
        }
      }).submitStudentExam(payload);
    }

    // Anti Cheat System
    function setupAntiCheat() {
      document.addEventListener("visibilitychange", () => {
        if (document.hidden && studentInfo) {
          cheatViolationCount++;
          document.getElementById('cheat-count').innerText = cheatViolationCount;
          document.getElementById('modal-cheat').classList.remove('hidden');
        }
      });
    }

    function closeCheatModal() {
      document.getElementById('modal-cheat').classList.add('hidden');
    }

    function blockDevToolsAndCopy() {
      document.addEventListener("copy", e => e.preventDefault());
      document.addEventListener("cut", e => e.preventDefault());
      document.addEventListener("paste", e => e.preventDefault());

      document.addEventListener("keydown", e => {
        if (
          e.key === "F12" || 
          (e.ctrlKey && e.shiftKey && (e.key === "I" || e.key === "J" || e.key === "C")) ||
          (e.ctrlKey && e.key === "U")
        ) {
          e.preventDefault();
        }
      });
    }

    // ==========================================
    // ADMIN DASHBOARD CLIENT LOGIC
    // ==========================================
    function switchAdminTab(tab) {
      const tabInput = document.getElementById('admin-tab-input');
      const tabRekap = document.getElementById('admin-tab-rekap');
      const secInput = document.getElementById('admin-sec-input');
      const secRekap = document.getElementById('admin-sec-rekap');

      if (tab === 'input') {
        tabInput.className = "py-3 text-sm font-bold text-emerald-600 border-b-2 border-emerald-600 focus:outline-none";
        tabRekap.className = "py-3 text-sm font-bold text-slate-500 hover:text-emerald-600 border-b-2 border-transparent focus:outline-none";
        secInput.classList.remove('hidden');
        secRekap.classList.add('hidden');
      } else {
        tabRekap.className = "py-3 text-sm font-bold text-emerald-600 border-b-2 border-emerald-600 focus:outline-none";
        tabInput.className = "py-3 text-sm font-bold text-slate-500 hover:text-emerald-600 border-b-2 border-transparent focus:outline-none";
        secRekap.classList.remove('hidden');
        secInput.classList.add('hidden');
        loadAdminResults();
      }
    }

    function updateLivePreview() {
      const no = document.getElementById('adm-no').value;
      const bobot = document.getElementById('adm-bobot').value;
      const stimulus = document.getElementById('adm-stimulus').value;
      const soal = document.getElementById('adm-soal').value;
      const optA = document.getElementById('adm-opt-a').value;
      const optB = document.getElementById('adm-opt-b').value;
      const optC = document.getElementById('adm-opt-c').value;
      const optD = document.getElementById('adm-opt-d').value;

      document.getElementById('prev-no').innerText = no || "1";
      document.getElementById('prev-bobot').innerText = bobot || "1";
      document.getElementById('prev-soal-text').innerText = soal || "Teks Pertanyaan akan muncul di sini...";
      document.getElementById('prev-opt-a').innerText = optA || "-";
      document.getElementById('prev-opt-b').innerText = optB || "-";
      document.getElementById('prev-opt-c').innerText = optC || "-";
      document.getElementById('prev-opt-d').innerText = optD || "-";

      const stimBox = document.getElementById('prev-stimulus-box');
      if (stimulus.trim()) {
        document.getElementById('prev-stimulus-text').innerText = stimulus;
        stimBox.classList.remove('hidden');
      } else {
        stimBox.classList.add('hidden');
      }
    }

    function saveAdminQuestion() {
      const qData = {
        nomor: document.getElementById('adm-no').value,
        jenis: document.getElementById('adm-jenis').value,
        bobot: document.getElementById('adm-bobot').value,
        teksStimulus: document.getElementById('adm-stimulus').value,
        teksSoal: document.getElementById('adm-soal').value,
        pilihanA: document.getElementById('adm-opt-a').value,
        pilihanB: document.getElementById('adm-opt-b').value,
        pilihanC: document.getElementById('adm-opt-c').value,
        pilihanD: document.getElementById('adm-opt-d').value,
        kunciJawaban: document.getElementById('adm-kunci').value,
        pembahasan: document.getElementById('adm-pembahasan').value
      };

      google.script.run.withSuccessHandler(res => {
        alert(res.message);
      }).saveSingleQuestion(qData);
    }

    function loadAdminResults() {
      google.script.run.withSuccessHandler(data => {
        const tbody = document.getElementById('tbl-rekap-body');
        if (!data || data.length === 0) {
          tbody.innerHTML = `<tr><td colspan="7" class="p-4 text-center text-slate-400">Belum ada hasil ujian.</td></tr>`;
          return;
        }

        tbody.innerHTML = data.map(r => `
          <tr class="hover:bg-slate-50 border-b">
            <td class="p-3 border">${r.timestamp}</td>
            <td class="p-3 border font-semibold text-slate-800">${r.nama}</td>
            <td class="p-3 border">${r.kelas}</td>
            <td class="p-3 border text-center font-bold text-emerald-600 text-sm">${r.nilaiAkhir}</td>
            <td class="p-3 border text-center text-emerald-600 font-semibold">${r.totalBenar}</td>
            <td class="p-3 border text-center text-rose-600 font-semibold">${r.totalSalah}</td>
            <td class="p-3 border text-center ${r.pelanggaranTab > 0 ? 'text-rose-600 font-bold bg-rose-50' : ''}">${r.pelanggaranTab} Kali</td>
          </tr>
        `).join('');
      }).getExamResultsForAdmin();
    }

    function logoutAdmin() {
      window.location.reload();
    }
  </script>
